        DWORD WINAPI HookedXInputGetState(DWORD idx, XINPUT_STATE* state) noexcept
        {
            const DWORD r = g_origXInput ? g_origXInput(idx, state) : ERROR_DEVICE_NOT_CONNECTED;
            CaptureLeftStick(state, r, idx);
            DetectRecenterCombo(state, r, idx);
            MELEVR::XrInput::LogRealPadThrottled(state, r);   // Stage 1: mapping discovery (T4), no-op unless cfg on
            bool synthesized = false;
            if (state != nullptr)
            {
                if (g_open)
                {
                    ZeroMemory(&state->Gamepad, sizeof(XINPUT_GAMEPAD));  // neutral while menu open
                }
                else
                {
                    const auto& c = MELEVR::Config::Get();
                    // Stage 1: virtual gamepad from the VR controllers. The synthesized
                    // state then passes through the SAME corrections as the real pad
                    // (decouple axes + move-follows-head stick rotation) - no duplication.
                    if (c.controllerInput && MELEVR::XrInput::BuildVirtualGamepad(state, c))
                    {
                        synthesized = true;
                        state->dwPacketNumber++;
                    }
                    const bool menuUi = MenuOwnsLook();
                    if (c.decoupledPitch && !menuUi) state->Gamepad.sThumbRY = 0;
                    if (c.decoupledYaw   && !menuUi) state->Gamepad.sThumbRX = 0;
                    if (c.moveFollowsHead) RotateMoveStickByHeadLook(&state->Gamepad);
                }
            }
            // Stage 1: while synthesizing the game must see a CONNECTED pad even when no
            // physical pad is plugged in (r can be ERROR_DEVICE_NOT_CONNECTED).
            return synthesized ? ERROR_SUCCESS : r;
        }

    HookedXInputGetStateEx - идентичные правки (та же логика, свой read trampoline).

    НЮАНС: BuildVirtualGamepad читает g_frame, опубликованный XrInput::OnFrame в том же
    кадре (render thread, до Present) - порядок гарантирован фрейм-лупом (пункт B из
    02_xr_session.cpp.txt идёт в OnPresent до отрисовки; XInput-поллинг игры происходит
    в game-тике, и worst case - стейт на кадр старше. Для aim это неважно (aim пишется
    из того же OnFrame), для триггера задержка на 1 кадр = норма VR-ввода.
